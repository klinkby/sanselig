{{- /* Renders a with a WebP source plus a fallback in the original format, both generated by Hugo's image pipeline (requires the extended build — see AGENTS.md). Params: path (required) resource path under assets/, e.g. "img/hero.jpg" alt (required, may be "") class (optional) class on the widths (optional) slice of ints — when set, a responsive srcset is built at each width for both formats sizes (optional) sizes attribute, required when widths is set loading, decoding, fetchpriority (optional) passed straight through */ -}} {{- $path := .path -}} {{- $img := resources.Get $path -}} {{- if not $img }}{{ errorf "picture.html: resource %q not found" $path }}{{ end -}} {{- $widths := .widths -}} {{- $class := .class -}} {{- $loading := .loading -}} {{- $decoding := .decoding -}} {{- $fetchpriority := .fetchpriority -}} {{- if $widths -}} {{- $sizes := .sizes -}} {{- $webpSet := slice -}} {{- $origSet := slice -}} {{- $fallback := $img -}} {{- range $widths -}} {{- $w := . -}} {{- $webp := $img.Resize (printf "%dx webp q75" $w) -}} {{- $webpSet = $webpSet | append (printf "%s %dw" $webp.RelPermalink $w) -}} {{- $orig := $img.Resize (printf "%dx q75" $w) -}} {{- $origSet = $origSet | append (printf "%s %dw" $orig.RelPermalink $w) -}} {{- if eq $w (index $widths (sub (len $widths) 1)) }}{{ $fallback = $orig }}{{ end -}} {{- end -}} {{- else -}} {{- $webp := $img.Resize (printf "%dx webp q75" $img.Width) -}} {{- end -}}
