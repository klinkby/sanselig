{{- /* Renders a with a WebP source plus a fallback in the original format, both generated by Hugo's image pipeline (requires the extended build — see AGENTS.md). Params: path (required) resource path under assets/, e.g. "img/hero.jpg" alt (required, may be "") class (optional) class on the widths (optional) slice of ints — when set, a responsive srcset is built at each width for both formats; otherwise a single native-size WebP + original is rendered sizes (optional) sizes attribute, only meaningful together with widths loading, decoding, fetchpriority (optional) passed straight through */ -}} {{- $path := .path -}} {{- $img := resources.Get $path -}} {{- if not $img }}{{ errorf "picture.html: resource %q not found" $path }}{{ end -}} {{- $widths := .widths | default (slice $img.Width) -}} {{- $sizes := .sizes -}} {{- $class := .class -}} {{- $loading := .loading -}} {{- $decoding := .decoding -}} {{- $fetchpriority := .fetchpriority -}} {{- $webpSet := partial "webp-srcset.html" (dict "path" $path "widths" $widths) -}} {{- $origSet := slice -}} {{- range $widths -}} {{- $orig := $img.Resize (printf "%dx q75" .) -}} {{- $origSet = $origSet | append (printf "%s %dw" $orig.RelPermalink .) -}} {{- end -}} {{- $fallback := $img.Resize (printf "%dx q75" (index $widths (sub (len $widths) 1))) -}}
